\newcommand{\partbackground}{%
  \AddToShipoutPictureBG*{%
    \begin{tikzpicture}[remember picture,overlay]
      \fill[navy] (current page.south west) rectangle (current page.north east);
      \begin{scope}[shift={(current page.south west)}]
        \foreach \a/\o in {0.55/0.55,0.4/0.35,0.25/0.22}{
          \draw[gold,opacity=\o,line width=0.8pt,domain=0:15.24,samples=220,smooth]
            plot (\x,{3.0+\a*3*exp(-0.12*\x)*sin(150*\x)});}
        \draw[white,opacity=0.18,line width=0.5pt] (0,3.0) -- (15.24,3.0);
      \end{scope}
    \end{tikzpicture}}}
